Polyglot Zero-Copy Demonstration

See how Alloy bridges JavaScript web routing directly to Python NumPy models over shared RAM:

import { http, memory } from 'alloy:core';
import { processTensor } from './ai_model.py' as python;

http.createServer(async (req, res) => {
  // 1. Allocate Float32 tensor directly into OS shared memory segment
  const sharedTensor = memory.allocateFloat32Array(req.body.imagePixels);

  // 2. Call Python directly — passes raw memory pointer (0-copy, 0.003ms)
  const prediction = await python.processTensor(sharedTensor.ptr, sharedTensor.length);

  // 3. Arena detonates local memory at response completion
  res.json({ label: prediction.category, confidence: prediction.score });
}).listen(8080);
import numpy as np

def processTensor(ptr, length):
    # Read raw pointer directly into NumPy array without memory allocation
    arr = np.ctypeslib.as_array(ptr, shape=(length,))
    normalized = (arr - np.mean(arr)) / np.std(arr)
    return {
        "category": "fraud_risk",
        "score": float(np.max(normalized))
    }
